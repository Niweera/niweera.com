(async function () {
  let manifest = [];
  try {
    const r = await fetch("/assets/data/vault.json", { cache: "no-cache" });
    if (r.ok) manifest = await r.json();
  } catch (_) {}

  const raw = location.pathname || "";
  let path = "";
  try {
    path = decodeURIComponent(raw).normalize("NFC").replace(/^\/+|\/+$/g, "");
  } catch (_) {
    path = raw.normalize("NFC").replace(/^\/+|\/+$/g, "");
  }
  if (!path || !Array.isArray(manifest) || manifest.length === 0) return;

  const enc = new TextEncoder();

  const hexToBytes = (h) => {
    const a = new Uint8Array(h.length / 2);
    for (let i = 0; i < a.length; i++) a[i] = parseInt(h.substr(i * 2, 2), 16);
    return a;
  };
  const b64ToBytes = (b) => {
    const s = atob(b);
    const a = new Uint8Array(s.length);
    for (let i = 0; i < s.length; i++) a[i] = s.charCodeAt(i);
    return a;
  };

  const km = await crypto.subtle.importKey(
    "raw", enc.encode(path), "PBKDF2", false, ["deriveKey"]
  );

  for (const entry of manifest) {
    try {
      const key = await crypto.subtle.deriveKey(
        { name: "PBKDF2", salt: hexToBytes(entry.salt), iterations: 250000, hash: "SHA-256" },
        km,
        { name: "AES-GCM", length: 256 },
        false,
        ["decrypt"]
      );
      const pt = await crypto.subtle.decrypt(
        { name: "AES-GCM", iv: hexToBytes(entry.iv) },
        key,
        b64ToBytes(entry.ct)
      );
      const html = new TextDecoder().decode(pt);
      const parsed = new DOMParser().parseFromString(html, "text/html");
      const out = document.getElementById("vault-output");
      out.replaceChildren(...parsed.body.childNodes);
      out.style.display = "block";
      document.getElementById("vault-fallback").style.display = "none";
      return;
    } catch (_) {
      continue;
    }
  }
})();
</script>
